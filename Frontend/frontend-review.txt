import axios from "axios";

// Set VITE_API_URL in Frontend/.env to change (e.g. for deployment)
export const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL || "http://localhost:5000/api",
});

// Reviewer name (no login in this app): sent with every request so verifications
// and manual additions are attributed in the audit trail.
const REVIEWER_KEY = "bgv.reviewerName";
export const getReviewerName = () => {
  try { return localStorage.getItem(REVIEWER_KEY) || ""; } catch { return ""; }
};
export const setReviewerName = (name) => {
  try { localStorage.setItem(REVIEWER_KEY, String(name || "").trim()); } catch { /* storage unavailable */ }
};
api.interceptors.request.use((config) => {
  const name = getReviewerName();
  if (name) config.headers["X-Reviewer"] = name;
  return config;
});

// Pull a readable message out of any axios error
export const errMsg = (e) =>
  e?.response?.data?.error ||
  e?.response?.data?.message ||
  e?.message ||
  "Something went wrong";

// ---- Candidates
export const getCandidates = () => api.get("/candidates").then((r) => r.data.data);
export const getCandidate = (id) => api.get(`/candidates/${id}`).then((r) => r.data.data);
export const createCandidate = (body) => api.post("/candidates", body).then((r) => r.data.data);
export const deleteCandidate = (id) => api.delete(`/candidates/${id}`);

// ---- BGV runs
export const getRuns = () => api.get("/bgv-runs").then((r) => r.data.data);
export const getRun = (id) => api.get(`/bgv-runs/${id}`).then((r) => r.data.data);
export const createRun = (candidateId) =>
  api.post("/bgv-runs", { candidateId }).then((r) => r.data.data);
// search + match can take a while (external search APIs)
export const searchRun = (id) =>
  api.post(`/bgv-runs/${id}/search`, {}, { timeout: 5 * 60 * 1000 }).then((r) => r.data.data);
export const matchRun = (id) =>
  api.post(`/bgv-runs/${id}/match`, {}, { timeout: 5 * 60 * 1000 }).then((r) => r.data.data);

// ---- Matches (human review)
export const getRelevantMatches = (runId, limit = 15, { view = "shortlist", page = 1 } = {}) =>
  api.get(`/candidate-matches/run/${runId}/relevant`, { params: { limit, view, page } }).then((r) => r.data);
export const reviewMatch = (id, reviewStatus, reviewerNotes) =>
  api.patch(`/candidate-matches/${id}`, { reviewStatus, reviewerNotes }).then((r) => r.data.data);

// ---- Per-platform discovery (each platform has its own target and counts)
export const getRunPlatforms = (runId) =>
  api.get(`/bgv-runs/${runId}/platforms`).then((r) => r.data.data);
export const getPlatformMatches = (runId, platform, { view = "shortlist", page = 1, limit } = {}) =>
  api.get(`/candidate-matches/run/${runId}/platform/${platform}`, { params: { view, page, ...(limit ? { limit } : {}) } }).then((r) => r.data);

// ---- Permanent verified identities + manual "Add profile"
export const getKnownProfiles = (runId) =>
  api.get(`/bgv-runs/${runId}/known-profiles`).then((r) => r.data.data);
// Resolves { ok, status, data, duplicate?, existing?, message }. A duplicate (409)
// is an expected answer, not an exception, so the UI can offer to open it.
export const addProfile = async (runId, body) => {
  try {
    const r = await api.post(`/bgv-runs/${runId}/profiles`, body);
    return { ok: true, status: r.status, ...r.data };
  } catch (e) {
    const d = e?.response?.data;
    if (e?.response?.status === 409 && d?.duplicate) return { ok: false, status: 409, ...d };
    throw e;
  }
};
export const revokeVerifiedIdentity = (id, reason) =>
  api.post(`/verified-identities/${id}/revoke`, { reason }).then((r) => r.data.data);

export const updateCandidate = (id, body) => api.patch(`/candidates/${id}`, body).then((r) => r.data.data);

// ---- Google Sheet import (review -> confirm -> sequential search)
export const getSheetStatus = () => api.get("/sheet-import/status").then((r) => r.data.data);
export const syncSheet = () => api.post("/sheet-import/sync", {}, { timeout: 2 * 60 * 1000 }).then((r) => r.data.data);
export const getSheetItems = () => api.get("/sheet-import/items").then((r) => r.data.data);
export const confirmSheetCandidates = (candidateIds, acknowledgeIncomplete = false) =>
  api.post("/sheet-import/confirm", { candidateIds, acknowledgeIncomplete }).then((r) => r.data.data);
export const unconfirmSheetCandidates = (candidateIds) =>
  api.post("/sheet-import/unconfirm", { candidateIds }).then((r) => r.data.data);
export const startSheetQueue = (candidateIds) => api.post("/sheet-import/queue/start", { candidateIds }).then((r) => r.data.data);
export const retrySheetFailed = (candidateIds) => api.post("/sheet-import/queue/retry", { candidateIds }).then((r) => r.data.data);
export const cancelSheetQueue = (candidateIds) => api.post("/sheet-import/queue/cancel", { candidateIds }).then((r) => r.data.data);

import { useCallback, useEffect, useMemo, useState } from "react";
import { Link } from "react-router-dom";
import {
  cancelSheetQueue, confirmSheetCandidates, errMsg, getSheetItems, getSheetStatus,
  retrySheetFailed, startSheetQueue, syncSheet, unconfirmSheetCandidates,
} from "../api/client";
import { Badge, ErrorBox, Loading, PageHeader, fmtDate } from "../components/ui";

const LEVEL = { COMPLETE: "Complete", SEARCHABLE: "Searchable (some details missing)", INCOMPLETE: "Incomplete" };
const QUEUE = { REVIEW: "Waiting for review", QUEUED: "In queue", RUNNING: "Searching nowâ€¦", DONE: "Search done", FAILED: "Search failed" };

export default function SheetImport() {
  const [status, setStatus] = useState(null);
  const [items, setItems] = useState(null);
  const [error, setError] = useState("");
  const [notice, setNotice] = useState("");
  const [busy, setBusy] = useState("");
  const [sel, setSel] = useState(new Set());

  const load = useCallback(async () => {
    try {
      const [s, i] = await Promise.all([getSheetStatus(), getSheetItems()]);
      setStatus(s); setItems(i); setError("");
    } catch (e) { setError(errMsg(e)); }
  }, []);
  useEffect(() => { load(); }, [load]);

  const active = (items || []).some((i) => ["QUEUED", "RUNNING"].includes(i.queueStatus));
  useEffect(() => {
    if (!active && !status?.syncing) return undefined;
    const t = setInterval(load, 4000);
    return () => clearInterval(t);
  }, [active, status?.syncing, load]);

  const act = async (name, fn) => {
    setBusy(name); setNotice(""); setError("");
    try { await fn(); await load(); } catch (e) { setError(errMsg(e)); }
    setBusy("");
  };

  const reviewable = useMemo(() => (items || []).filter((i) => i.candidateId && i.queueStatus === "REVIEW"), [items]);
  const chosen = useMemo(() => (items || []).filter((i) => sel.has(i.candidateId)), [items, sel]);
  const chosenReview = chosen.filter((i) => i.queueStatus === "REVIEW");
  const chosenIncomplete = chosenReview.filter((i) => i.completeness?.level === "INCOMPLETE" || i.completeness?.invalid?.length);
  const readyConfirmed = (items || []).filter((i) => i.queueStatus === "REVIEW" && i.confirmed);
  const failed = (items || []).filter((i) => i.queueStatus === "FAILED");
  const queued = (items || []).filter((i) => i.queueStatus === "QUEUED");

  const toggle = (id) => setSel((s) => { const n = new Set(s); n.has(id) ? n.delete(id) : n.add(id); return n; });
  const selectAllReview = () => setSel(new Set(reviewable.map((i) => i.candidateId)));

  const doSync = () => act("sync", async () => {
    const r = await syncSheet();
    const skipped = r.skipped?.length ? `, ${r.skipped.length} row(s) skipped (${r.skipped.slice(0, 3).map((x) => `row ${x.row}: ${x.reason}`).join("; ")})` : "";
    setNotice(`Synced: ${r.imported} new, ${r.updated} updated, ${r.alreadyImported} already imported, ${r.linkedExisting} matched existing${skipped}. Nothing was searched.`);
  });

  const confirm = (acknowledge) => act("confirm", async () => {
    const ids = [...sel];
    const r = await confirmSheetCandidates(ids, acknowledge);
    setNotice(`${r.confirmed.length} confirmed.${r.needsAcknowledgement.length ? ` ${r.needsAcknowledgement.length} incomplete/invalid not confirmed â€” use "Search with incomplete data" if you really want them.` : ""}`);
  });

  const confirmIncomplete = () => {
    const names = chosenIncomplete.map((i) => i.name).join(", ");
    if (!window.confirm(`Search with incomplete data?\n\n${names}\n\nThese candidates have missing or invalid details. Results may be weaker or less certain. Continue?`)) return;
    confirm(true);
  };

  const start = () => {
    if (!window.confirm(`Start the sequential search for ${readyConfirmed.length} confirmed candidate(s)? They are searched one at a time and this uses your search providers.`)) return;
    act("start", async () => {
      const r = await startSheetQueue();
      setNotice(`${r.queued} candidate(s) queued. They are searched one at a time.`);
    });
  };

  if (!status && !error) return <Loading />;
  const q = status?.queue;
  const doneCount = q ? q.done : 0;

  return (
    <>
      <PageHeader title="Google Sheet" subtitle="Rows are imported automatically; nothing is searched until you confirm.">
        <button className="btn" onClick={doSync} disabled={busy === "sync" || !status?.configured}>{busy === "sync" ? "Syncingâ€¦" : "âŸ³ Sync now"}</button>
      </PageHeader>

      {error && <ErrorBox message={error} onRetry={load} />}
      {notice && <div className="state">{notice}</div>}

      {status && (
        <div className="card">
          <h3>Connection</h3>
          {!status.configured ? (
            <div className="state error">Not connected: {status.problem}. See the setup steps in the README / .env.example.</div>
          ) : (
            <>
              <div>Connected Â· tab <b>{status.tab || "(first tab)"}</b> Â· checked every {status.pollingMinutes > 0 ? `${status.pollingMinutes} min` : "â€” (manual only)"} (polling, not instant)</div>
              <div className="muted">Last successful sync: {status.lastSuccessAt ? fmtDate(status.lastSuccessAt) : "never"} Â· Searching is {status.searchEnabled ? "enabled" : "DISABLED"} Â· one-by-one, {Math.round(status.queueDelayMs / 100) / 10}s between candidates</div>
              {status.lastError && <div className="state error">Last sync problem: {status.lastError}</div>}
            </>
          )}
        </div>
      )}

      {q && q.review > 0 && (
        <div className="state">Candidate data imported. Please review completeness before starting searches.</div>
      )}

      {q && q.total > 0 && (
        <div className="card">
          <h3>Search queue</h3>
          <div>
            {q.review} to review ({q.reviewConfirmed} confirmed) Â· {q.queued} waiting Â· {q.running} running Â· {doneCount} done Â· {q.failed} failed
          </div>
          {q.current?.length > 0 && <div>Now searching: <b>{q.current.map((c) => c.name).join(", ")}</b>{queued.length ? ` Â· ${queued.length} waiting after it` : ""}</div>}
          <progress max={q.total || 1} value={doneCount} style={{ width: "100%" }} />
          <div style={{ marginTop: 8, display: "flex", gap: 8, flexWrap: "wrap" }}>
            <button className="btn primary" disabled={!readyConfirmed.length || !status?.searchEnabled || busy === "start"} onClick={start}>
              Start sequential search ({readyConfirmed.length})
            </button>
            {failed.length > 0 && <button className="btn" disabled={busy === "retry"} onClick={() => act("retry", async () => { const r = await retrySheetFailed(); setNotice(`${r.retried} re-queued.`); })}>Retry failed ({failed.length})</button>}
            {queued.length > 0 && <button className="btn" onClick={() => act("cancel", async () => { const r = await cancelSheetQueue(); setNotice(`${r.cancelled} taken out of the queue.`); })}>Cancel waiting ({queued.length})</button>}
          </div>
        </div>
      )}

      <div className="card flush">
        <div style={{ padding: 12, display: "flex", gap: 8, flexWrap: "wrap", alignItems: "center" }}>
          <button className="btn small" onClick={selectAllReview}>Select all awaiting review</button>
          <button className="btn small" onClick={() => setSel(new Set())}>Clear</button>
          <span className="muted">{chosen.length} selected</span>
          <button className="btn small primary" disabled={!chosenReview.length || busy === "confirm"} onClick={() => confirm(false)}>Confirm data is complete</button>
          <button className="btn small" disabled={!chosenIncomplete.length || busy === "confirm"} onClick={confirmIncomplete}>Search with incomplete dataâ€¦</button>
          <button className="btn small" disabled={!chosenReview.length} onClick={() => act("unconfirm", async () => { await unconfirmSheetCandidates([...sel]); })}>Un-confirm</button>
        </div>
        {!items ? <Loading /> : (
          <table>
            <thead><tr><th /><th>Row / Job</th><th>Candidate</th><th>Import</th><th>Completeness</th><th>Missing / invalid</th><th>Search</th><th /></tr></thead>
            <tbody>
              {items.map((i) => (
                <tr key={i.id}>
                  <td><input type="checkbox" disabled={i.queueStatus !== "REVIEW" || !i.candidateId} checked={sel.has(i.candidateId)} onChange={() => toggle(i.candidateId)} /></td>
                  <td>#{i.rowNumber}<div className="muted">{i.jobId || "â€”"}</div></td>
                  <td>{i.candidateId ? <Link to={`/candidates/${i.candidateId}`}>{i.name}</Link> : "â€”"}<div className="muted">{i.email}</div></td>
                  <td><Badge value={i.importStatus} />{i.duplicateReason && <div className="muted">{i.duplicateReason}</div>}{i.sourceDuplicateStatus && <div className="muted">sheet: {i.sourceDuplicateStatus}</div>}</td>
                  <td>{i.completeness ? <><Badge value={i.completeness.level} /><div className="muted">{LEVEL[i.completeness.level]}</div></> : "â€”"}</td>
                  <td style={{ maxWidth: 260 }}>
                    {i.completeness?.missing?.length > 0 && <div>Missing: {i.completeness.missing.join(", ")}</div>}
                    {i.completeness?.invalid?.length > 0 && <div className="muted">Invalid: {i.completeness.invalid.map((x) => `${x.field} (${x.reason})`).join("; ")}</div>}
                    {i.completeness?.identifiers?.length > 0 && <div className="muted">Can search by: {i.completeness.identifiers.join(", ")}</div>}
                    {i.candidateId && <Link className="btn small" to={`/candidates/${i.candidateId}`}>Edit details</Link>}
                  </td>
                  <td>
                    <Badge value={i.queueStatus} />
                    <div className="muted">{QUEUE[i.queueStatus]}{i.queuePosition ? ` Â· #${i.queuePosition}` : ""}{i.confirmed && i.queueStatus === "REVIEW" ? " Â· confirmed" : ""}{i.incompleteOverride ? " Â· incomplete override" : ""}</div>
                    {i.error && <div className="state error" style={{ padding: 4 }}>{i.error}</div>}
                  </td>
                  <td className="right">{i.bgvRun && <Link className="btn small" to={`/runs/${i.bgvRun}`}>Open results</Link>}</td>
                </tr>
              ))}
              {items.length === 0 && <tr><td colSpan="8" className="muted">No rows imported yet. Press â€œSync nowâ€.</td></tr>}
            </tbody>
          </table>
        )}
      </div>
    </>
  );
}

========== src\pages\Candidates.jsx ==========

import { useMemo, useState } from "react";
import { Link, useNavigate } from "react-router-dom";
import { createRun, deleteCandidate, errMsg, getCandidates } from "../api/client";
import useAsync from "../components/useAsync";
import { Badge, ErrorBox, Loading, PageHeader } from "../components/ui";

export default function Candidates() {
  const nav = useNavigate();
  const { data, loading, error, reload } = useAsync(getCandidates);
  const [q, setQ] = useState("");
  const [busy, setBusy] = useState(null);
  const [actionError, setActionError] = useState("");

  const rows = useMemo(() => {
    const s = q.trim().toLowerCase();
    if (!data) return [];
    return s
      ? data.filter((c) => [c.fullName, c.email, c.currentCompany].some((v) => v?.toLowerCase().includes(s)))
      : data;
  }, [data, q]);

  const startBgv = async (c) => {
    setBusy(c._id); setActionError("");
    try {
      const run = await createRun(c._id);
      nav(`/runs/${run._id}`);
    } catch (e) { setActionError(errMsg(e)); setBusy(null); }
  };

  const remove = async (c) => {
    if (!window.confirm(`Delete ${c.fullName}?`)) return;
    try { await deleteCandidate(c._id); reload(); } catch (e) { setActionError(errMsg(e)); }
  };

  return (
    <>
      <PageHeader title="Candidates" subtitle={data ? `${data.length} total` : ""}>
        <input className="search" placeholder="Search name, email, companyâ€¦" value={q} onChange={(e) => setQ(e.target.value)} />
        <Link className="btn primary" to="/candidates/new">+ Add candidate</Link>
      </PageHeader>

      {actionError && <div className="state error">{actionError}</div>}
      {loading ? <Loading /> : error ? <ErrorBox message={error} onRetry={reload} /> : (
        <div className="card flush">
          <table>
            <thead><tr><th>Name</th><th>Current role</th><th>Contact</th><th>Status</th><th /></tr></thead>
            <tbody>
              {rows.map((c) => (
                <tr key={c._id}>
                  <td><Link to={`/candidates/${c._id}`}><b>{c.fullName}</b></Link></td>
                  <td>{[c.currentJobTitle, c.currentCompany].filter(Boolean).join(" Â· ") || "â€”"}</td>
                  <td className="muted">{c.email || c.phone || "â€”"}</td>
                  <td><Badge value={c.verificationStatus} /></td>
                  <td className="right nowrap">
                    <button className="btn small primary" disabled={busy === c._id} onClick={() => startBgv(c)}>
                      {busy === c._id ? "Startingâ€¦" : "Start BGV"}
                    </button>{" "}
                    <button className="btn small danger" onClick={() => remove(c)}>Delete</button>
                  </td>
                </tr>
              ))}
              {rows.length === 0 && <tr><td colSpan="5" className="muted">No candidates found.</td></tr>}
            </tbody>
          </table>
        </div>
      )}
    </>
  );
}
